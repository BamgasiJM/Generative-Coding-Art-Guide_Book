- 숫자/배열/외부파일(csv/json/api) 등을 형태 생성 알고리즘의 입력으로 사용

```javascript
// === Data-driven Geometry ===
// CSV 데이터 → Radial Sculpture 형태로 변환
// 데이터 기반 기하 조형

let table;
let values = [];

function preload() {
  // CSV 파일을 불러옴
  table = loadTable("data.csv", "csv", "header");
}

function setup() {
  createCanvas(700, 700);
  angleMode(DEGREES);

  // CSV에서 데이터 추출
  for (let r = 0; r < table.getRowCount(); r++) {
    let v = table.getNum(r, "value");
    values.push(v);
  }
}

function draw() {
  background(20);
  translate(width / 2, height / 2);

  // 시각화 파라미터
  let count = values.length;
  let angleStep = 360 / count;

  for (let i = 0; i < count; i++) {
    let v = values[i];

    // 데이터 → Geometry 변환
    let radius = map(v, 0, 100, 40, 300);
    let thickness = map(v, 0, 100, 4, 20);
    let hue = map(v, 0, 100, 160, 360);

    push();
    rotate(i * angleStep);

    // 색상
    fill(hue, 150, 255);
    noStroke();

    // 방사형 기둥 (Radial Bar)
    rect(0, -thickness / 2, radius, thickness, 5);

    // 데이터 점 (점토 조형 느낌)
    fill(255);
    circle(radius, 0, thickness * 0.8);

    pop();
  }
}
```
