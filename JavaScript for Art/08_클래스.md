## 개념 설명

클래스는 연관된 \*\*데이터(변수)\*\*와 \*\*행동(함수)\*\*을 하나의 캡슐로 묶어내는 설계도입니다. p5.js나 Three.js 작업은 결국 수많은 "**객체들(Objects)**"의 상호작용입니다. 클래스를 사용하면 코드를 "부품화"하여 재사용하고 확장하기 쉬워집니다. `constructor`(생성자)는 객체가 처음 만들어질 때 실행되며 초기 상태를 설정합니다.

## p5.js 예시 : 자율 에이전트 설계

상황: 화면을 돌아다니는 수백 개의 'Walker'를 만듭니다. 각 Walker는 자신의 위치(`x`, `y`)와 색상(`color`)을 기억하고 스스로 움직여야 합니다.

```javascript
// 설계도(Class) 정의
class Walker {
  constructor(x, y) {
    // 'this'는 생성될 '나 자신'을 가리킵니다.
    this.pos = createVector(x, y);
    this.vel = p5.Vector.random2D();
    this.color = color(random(255), random(255), random(255));
  }

  update() {
    this.pos.add(this.vel);
    
    // 화면 밖으로 나가면 반대편으로
    if (this.pos.x > width) this.pos.x = 0;
    if (this.pos.x < 0) this.pos.x = width;
    if (this.pos.y > height) this.pos.y = 0;
    if (this.pos.y < 0) this.pos.y = height;
  }

  display() {
    fill(this.color);
    noStroke();
    ellipse(this.pos.x, this.pos.y, 10, 10);
  }
}

let walkers = [];

function setup() {
  createCanvas(400, 400);
  // Walker 인스턴스 50개 생성
  for (let i = 0; i < 50; i++) {
    walkers.push(new Walker(width / 2, height / 2));
  }
}

function draw() {
  background(220);
  for (let w of walkers) {
    w.update();
    w.display();
  }
}
```

## Three.js 예시 : 커스텀 3D 오브젝트 확장

상황: Three.js의 기본 `Mesh`나 `Group`을 \*\*상속(Extends)\*\*받아, 나만의 기능을 가진 로봇 클래스를 만듭니다. 이렇게 하면 `scene.add(myRobot)` 처럼 Three.js의 기본 기능과 완벽하게 호환됩니다.

```javascript
import * as THREE from 'three';

// THREE.Group을 상속받아 나만의 복합 객체를 만듭니다.
class Robot extends THREE.Group {
  constructor() {
    super(); // 부모(THREE.Group)의 생성자를 반드시 먼저 호출해야 합니다.

    // 1. 몸통
    const bodyGeo = new THREE.BoxGeometry(1, 2, 1);
    const bodyMat = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
    this.body = new THREE.Mesh(bodyGeo, bodyMat);
    this.add(this.body); // 'this'는 그룹 자체이므로 add 메서드를 가집니다.

    // 2. 머리
    const headGeo = new THREE.BoxGeometry(0.5, 0.5, 0.5);
    const headMat = new THREE.MeshBasicMaterial({ color: 0xff0000 });
    this.head = new THREE.Mesh(headGeo, headMat);
    this.head.position.y = 1.5;
    this.add(this.head);

    // 고유 데이터
    this.rotationSpeed = 0.01;
  }

  // 나만의 업데이트 함수
  update() {
    this.rotation.y += this.rotationSpeed; // 전체 회전
    this.head.rotation.y += 0.05; // 머리만 따로 회전
  }
}

// 사용하기
const myRobot = new Robot();
scene.add(myRobot); // THREE.Group을 상속받았기 때문에 바로 추가 가능!

function animate() {
  requestAnimationFrame(animate);
  myRobot.update(); // 매 프레임 호출
  renderer.render(scene, camera);
}
```
